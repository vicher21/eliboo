const categories = ['Todos', 'Peluches', 'Didácticos', 'Imaginación'];
const formatPrice = (price: number) => new Intl.NumberFormat('es-CO', { style: 'currency', currency: 'COP', maximumFractionDigits: 0 }).format(price);

export default function Home() {
  const [category, setCategory] = useState('Todos');
  const [search, setSearch] = useState('');
  const [cart, setCart] = useState<CartItem[]>([]);
  const [cartOpen, setCartOpen] = useState(false);

  const filteredProducts = useMemo(() => products.filter((product) => {
    const matchesCategory = category === 'Todos' || product.category === category;
    const matchesSearch = `${product.name} ${product.description} ${product.category}`.toLowerCase().includes(search.toLowerCase());
    return matchesCategory && matchesSearch;
  }), [category, search]);

  const cartCount = cart.reduce((sum, item) => sum + item.quantity, 0);
  const total = cart.reduce((sum, item) => sum + item.product.price * item.quantity, 0);

  function addToCart(product: Product) {
    setCart((current) => {
      const existing = current.find((item) => item.product.id === product.id);
      if (existing) return current.map((item) => item.product.id === product.id ? { ...item, quantity: item.quantity + 1 } : item);
      return [...current, { product, quantity: 1 }];
    });
    setCartOpen(true);
  }

  function changeQuantity(productId: number, change: number) {
    setCart((current) => current
      .map((item) => item.product.id === productId ? { ...item, quantity: item.quantity + change } : item)
      .filter((item) => item.quantity > 0));
  }

  function checkoutWhatsApp() {
    // Reemplaza este número por el WhatsApp de EliBoo, con indicativo de país y sin + ni espacios.
    const whatsappNumber = '57XXXXXXXXXX';
    const lines = cart.map(({ product, quantity }) => `• ${product.name} x${quantity} — ${formatPrice(product.price * quantity)}`);
    const message = `¡Hola, EliBoo! 🧸💗 Quiero hacer este pedido:\n\n${lines.join('\n')}\n\n*Total de productos: ${formatPrice(total)}*\n\n¿Me ayudan a confirmar disponibilidad y envío?`;
    if (whatsappNumber.includes('X')) {
      alert('Antes de recibir pedidos, abre app/page.tsx y cambia 57XXXXXXXXXX por el número de WhatsApp de tu tienda.');
      return;
    }
    window.open(`https://wa.me/${whatsappNumber}?text=${encodeURIComponent(message)}`, '_blank', 'noopener,noreferrer');
  }

    <main>
      <div className="announcement">Un mundo de ternura te espera <span>♡</span> Envíos por confirmar según tu ciudad</div>
      <header className="site-header">
        <a className="brand" href="#inicio" aria-label="EliBoo, inicio"><img src="/eliboo-logo.png" alt="Logo EliBoo: osito y letras pastel" /></a>

          <a href="#inicio">Inicio</a><a href="#catalogo">Catálogo</a><a href="#nosotros">Nosotros</a><a href="#contacto">Contacto</a>
        </nav>
        <button className="cart-trigger" onClick={() => setCartOpen(true)} aria-label={`Abrir carrito, ${cartCount} productos`}>

        </button>
      </header>

        <div className="hero-copy">
          <span className="eyebrow">✿ JUGUETES CON AMOR</span>

          <a className="button button-primary" href="#catalogo">Explorar catálogo <span>→</span></a>
          <div className="hero-note"><span>♡</span> Hecho para jugar, descubrir y soñar.</div>

          <div className="art-caption">Pequeños juguetes, grandes recuerdos</div>
        </div>

        <div className="benefit"><span className="benefit-icon peach">♧</span><div><strong>Compra acompañada</strong><small>Te ayudamos por WhatsApp</small></div></div>
      </section>

      <section className="catalog-section" id="catalogo">
        <div className="section-heading"><div><span className="eyebrow">NUESTROS FAVORITOS</span><h2>Un juguete para cada aventura <span>♡</span></h2><p>Encuentra ese detalle especial para llenar sus días de magia.</p></div><span className="product-count">{filteredProducts.length} productos</span></div>
        <div className="catalog-tools">
          <div className="category-pills" aria-label="Filtrar por categoría">{categories.map((item) => <button key={item} className={category === item ? 'pill active' : 'pill'} onClick={() => setCategory(item)}>{item}</button>)}</div>
          <label className="search-box"><span>⌕</span><input value={search} onChange={(event) => setSearch(event.target.value)} placeholder="Busca tu próximo favorito..." /></label>
        </div>
        <div className="product-grid">
          {filteredProducts.map((product) => <article className="product-card" key={product.id}>

          </article>)}
        </div>
        {filteredProducts.length === 0 && <div className="empty-state">No encontramos juguetes con esa búsqueda. Intenta con otra palabra 💗</div>}
      </section>

      <section className="love-banner" id="nosotros"><div className="love-sticker">♡</div><div><span className="eyebrow">BIENVENIDOS A ELIBOO</span><h2>La infancia está hecha de pequeños grandes momentos.</h2><p>Queremos ayudarte a encontrar juguetes que inspiren historias, despierten curiosidad y se conviertan en recuerdos bonitos.</p></div><a className="button button-light" href="#catalogo">Conocer juguetes →</a></section>

Hecho con amor.</small></footer>

            <div className="cart-items">{cart.map(({ product, quantity }) => <div className="cart-item" key={product.id}><img src={product.image} alt="" /><div className="cart-item-main"><strong>{product.name}</strong><b>{formatPrice(product.price)}</b><div className="quantity-control"><button onClick={() => changeQuantity(product.id, -1)} aria-label="Quitar uno">−</button><span>{quantity}</span><button onClick={() => changeQuantity(product.id, 1)} aria-label="Agregar uno">＋</button></div></div><button className="remove-item" onClick={() => setCart((current) => current.filter((item) => item.product.id !== product.id))} aria-label={`Eliminar ${product.name}`}>×</button></div>)}</div>
            <div className="cart-summary"><div><span>Subtotal</span><strong>{formatPrice(total)}</strong></div><p>El costo de envío se confirma por WhatsApp según tu ubicación.</p><div className="total-line"><span>Total de productos</span><strong>{formatPrice(total)}</strong></div><button className="whatsapp-button" onClick={checkoutWhatsApp}><span>☏</span> Finalizar pedido por WhatsApp</button><small>Tu pedido se confirmará directamente con EliBoo.</small></div>
          </>}
        </aside>
      </div>}
    </main>
  );
}
